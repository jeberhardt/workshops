let hue = 0;
const ws = new WebSocket("ws://localhost:8765");

ws.onmessage = (e) => {
  if (e.data === "press") hue = (hue + 60) % 360;
};

function setup() {
  createCanvas(windowWidth, windowHeight);
  colorMode(HSB);
}

function draw() {
  background(hue, 80, 90);
}

Each press shifts the colour. That’s the whole loop working end to end.
